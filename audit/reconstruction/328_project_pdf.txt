load_data <- function(csv_path) {
df = read.csv(csv_path)
df$has_y = as.factor(df$has_y)
df$tvae = as.factor(df$tvae)
df$method = as.factor(df$method)
df$approach = as.factor(df$approach)
df$dataset = as.factor(df$dataset)
df$y_synth = as.factor(df$y_synth)
return (df)
}

action terms involving the ctgan and tvae levels of y_synth are aliased. They are linearly dependent on their
corresponding main effects (y_synthctgan, y_synthtvae, tvae1). For example, when it is y_synthctgan, we
automatically know it is also has_y1 because has_y0 does not exist for y_synthctgan.

one_inter_boxcox <- function(df_nog, mod3) {
# --- Perform Box-Cox Transformation Analysis ---
cat("\n\n--- Performing Box-Cox Analysis ---\n")
# The boxcox function plots log-likelihood vs lambda and returns the values
# It requires the MASS package
bc_results <- boxcox(mod3, plotit = TRUE) # plotit=TRUE is default
# Find the lambda that maximizes the log-likelihood
optimal_lambda <- bc_results$x[which.max(bc_results$y)]
cat(paste("\nOptimal lambda found by boxcox:", optimal_lambda, "\n"))
# --- Apply Transformation and Refit Model ---
lambda_to_apply <- optimal_lambda
cat(paste("Applying transformation with lambda =", lambda_to_apply, "\n"))
14

PAGE 15
# Create the transformed response variable
if (abs(lambda_to_apply) < 1e-6) { # Check if lambda is effectively zero
df_nog$macro_max_transformed <- log(df_nog$macro_max)
transform_desc <- "log(macro_max)"
cat("Lambda is close to 0. Using log transformation.\n")
} else {
df_nog$macro_max_transformed <- (df_nog$macro_maxˆlambda_to_apply - 1) / lambda_to_apply
transform_desc <- paste0("(macro_maxˆ",round(lambda_to_apply, 3), " - 1) / ", round(lambda_to_apply, 3))
cat(paste("Using Box-Cox transformation with lambda =", lambda_to_apply, "\n"))
}
# --- Fit the Model with Transformed Response ---
cat("\n--- Fitting Model with Transformed Response ---\n")
# Create the formula string dynamically for clarity
formula_transformed_str <- paste(transform_desc, "~ y_synth * dataset + has_y + tvae")
cat("Formula:", formula_transformed_str, "\n")
# Fit the new model
# Use reformulate to handle the response string properly
mod3_transformed <- lm(reformulate(termlabels = c("y_synth * dataset", "has_y", "tvae"),
response = "macro_max_transformed"),
data = df_nog)
cat("\n--- Summary for Transformed Model ---\n")
print(summary(mod3_transformed))
cat("\n--- ANOVA Table (Type I) for Transformed Model ---\n")
print(anova(mod3_transformed))
cat("\n--- ANOVA Table (Type III) for Transformed Model ---\n")
# Use car::Anova for Type III SS
print(Anova(mod3_transformed, type="III"))
cat("\n--- Diagnostic Plots for Transformed Model ---\n")
# Check the diagnostic plots for the *new* model
par(mfrow = c(2, 2)) # Arrange plots in a 2x2 grid
plot(mod3_transformed)
par(mfrow = c(1, 1)) # Reset plotting layout
cat("\n")
# Return the transformed model (optional)
# return(list(original_model=mod3, transformed_model=mod3_transformed, lambda=lambda_to_apply))
invisible(mod3_transformed) # Return invisibly if called like: new_model <- one_inter_boxcox(df)
}
